			\includecpp{DFS}
			{Algoritmo de búsqueda que va lo más profundo posible en un grafo con $n$ nodos y $m$ aristas.}
			{$O(n + m)$}
			{\begin{itemize}
				\item Encontrar para un nodo $u$ algún camino que le permita llegar a cualquier otro nodo del grafo (si es posible)
        \item Ver si un nodo en un árbol es ancestro de otro haciendo uso de arreglos de tiempos de entrada y salida 
				\item Topological sort usando un arreglo para tiempos de salida
				\item Encontrar componentes fuertemente conexas
			\end{itemize}}
			{./Graphs}
			{DFS.cpp}
			
			\includecpp{BFS}
			{Algoritmo para recorrer grafos que visita los nodos en amplitud, análogo a la forma en la que se expande un incendio: si un nodo está ``ardiendo'', sus vecinos arden después de manera simultanea.}
			{$O(n + m)$}
			{\begin{itemize}
			  \item Encontrar el camino más corto en un grafo sin pesos 
        \item Encontrar todas las componentes conexas en un grafo 
        \item Encontrar el ciclo más corto conteniendo a un nodo origen en un grafo dirigido sin pesos
        \item Verificar si una arista $(u, v)$ pertenece a uno de los caminos más cortos entre $a$ y $b$ con $d_{a}[u] + 1 + d_{b}[v] = d_{a}[b]$
        \item Verificar si un nodo $v$ pertenece a uno de los caminos más cortos entre $a$ y $b$ con $d_{a}[v] + d_{b}[v] = d_{a}[b]$
			\end{itemize}}
			{./Graphs}
			{BFS.cpp}
			
			\includecpp{Dijkstra}
			{Algoritmo que permite encontrar las distancias más cortas de un nodo de inicio a todos los demás nodos en un grafo pesado. Los pesos del grafo deben ser todos positivos.}
			{$O(n^{2} + m)$}
			{Encontrar la distancia más corta de un nodo dado a todos los demás en un grafo.}
			{./Graphs}
